add_compile_definitions(PS3EYE_DEBUG)

if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/ps3eye-driver/CMakeLists.txt")
    # libusb must be located *before* add_subdirectory(). Directory properties
    # such as the link libraries and search paths are only inherited by
    # subdirectories added after the call, and ps3eye-driver links a bare
    # -lusb-1.0. Discovering libusb afterwards left the driver's own executables
    # without a search path, which breaks wherever libusb is not in a default
    # linker directory -- e.g. Homebrew on macOS:
    #     ld: library 'usb-1.0' not found
    #     FAILED: video-ps3eye/ps3eye-driver/ps3eye-frame-test
    if(NOT MSVC)
        find_package(PkgConfig QUIET)
        if(PKG_CONFIG_FOUND)
            pkg_check_modules(libusb "libusb-1.0" QUIET IMPORTED_TARGET)
        endif()
        if(libusb_FOUND)
            link_directories(${libusb_LIBRARY_DIRS})
            link_libraries(PkgConfig::libusb)
            if(WIN32)
                set(_bins "")
                foreach(_dir IN LISTS libusb_LIBRARY_DIRS)
                    list(APPEND _bins "${_dir}/../bin")
                endforeach()
                find_file(LIBUSB_DLL NAMES "libusb-1.0.dll" HINTS ${_bins} NO_DEFAULT_PATH)
                if(LIBUSB_DLL)
                    install(FILES "${LIBUSB_DLL}"
                            DESTINATION "${opentrack-libexec}"
                            PERMISSIONS ${opentrack-perms-exec})
                else()
                    otr_install_lib("${LIBUSB_DLL}" "${opentrack-libexec}")
                endif()
            endif()
        endif()
    else()
        set(SDK_LIBUSB CACHE PATH "")
        if(SDK_LIBUSB)
            set(libusb_FOUND TRUE)
            link_directories("${SDK_LIBUSB}")
            include_directories(SYSTEM "${SDK_LIBUSB}")
            set(path "${SDK_LIBUSB}/libusb-1.0.dll")
            if(EXISTS "${path}")
                otr_install_lib("${path}" "${opentrack-libexec}")
            endif()
            set(pdb "${SDK_LIBUSB}/libusb-1.0.pdb")
            install(FILES "${pdb}" DESTINATION "${opentrack-debug}" OPTIONAL)
        endif()
    endif()

    add_subdirectory("ps3eye-driver")
endif()

    if(WIN32)
        foreach(k ${SDL2_LIBRARIES})
            get_filename_component(path "${k}" PATH)
            set(lib "${path}/SDL2.dll")
            if(EXISTS "${lib}")
                otr_install_lib("${lib}" "${opentrack-libexec}")
                break()
            endif()
        endforeach()
    endif()
endif()
